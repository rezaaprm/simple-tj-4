import pymysql, math, heapq, time
from collections import defaultdict

DB_CONFIG = {
    'host': '127.0.0.1',
    'user': 'root',
    'password': '',
    'database': 'db_transportasi_jakarta_4'
}
TRANSFER_PENALTY = 2500
BUS_WEIGHT = 800
LONG_BUS_PENALTY = 100000
STAY_BONUS = 0
WALK_MULTIPLIER_SHORT = 5
WALK_MULTIPLIER_LONG = 50
MAX_WALK = 300

def haversine(lat1, lon1, lat2, lon2):
    R = 6371000
    phi1, phi2 = math.radians(lat1), math.radians(lat2)
    dphi = math.radians(lat2 - lat1)
    dlambda = math.radians(lon2 - lon1)
    a = math.sin(dphi/2)**2 + math.cos(phi1)*math.cos(phi2)*math.sin(dlambda/2)**2
    return R * 2 * math.atan2(math.sqrt(a), math.sqrt(1-a))

def fetch_stops():
    conn = pymysql.connect(**DB_CONFIG)
    cursor = conn.cursor(pymysql.cursors.DictCursor)
    cursor.execute("SELECT stop_id, stop_name, stop_lat, stop_lon FROM tb_stops")
    stops = cursor.fetchall()
    conn.close()
    return {s['stop_id']: {'name': s['stop_name'], 'lat': float(s['stop_lat']), 'lon': float(s['stop_lon'])} for s in stops}

def fetch_routes_with_stops():
    conn = pymysql.connect(**DB_CONFIG)
    cursor = conn.cursor(pymysql.cursors.DictCursor)
    cursor.execute("""
        SELECT t.route_id, st.stop_id, st.stop_sequence
        FROM tb_trips t JOIN tb_stop_times st ON t.trip_id = st.trip_id
        ORDER BY t.route_id, st.stop_sequence
    """)
    rows = cursor.fetchall()
    conn.close()
    routes = defaultdict(list)
    for row in rows:
        routes[row['route_id']].append(row['stop_id'])
    merged = {}
    for rid, stops in routes.items():
        seen = set()
        merged[rid] = [s for s in stops if not (s in seen or seen.add(s))]
    return merged

def build_graph(stop_map, routes):
    graph = defaultdict(list)
    for route_id, stops in routes.items():
        for i in range(len(stops)-1):
            s1, s2 = stops[i], stops[i+1]
            if s1 not in stop_map or s2 not in stop_map:
                continue
            lat1, lon1 = stop_map[s1]['lat'], stop_map[s1]['lon']
            lat2, lon2 = stop_map[s2]['lat'], stop_map[s2]['lon']
            dist = haversine(lat1, lon1, lat2, lon2)
            weight = dist + BUS_WEIGHT
            if dist > 4000:
                weight += LONG_BUS_PENALTY
            graph[s1].append({'to': s2, 'weight': weight, 'route_id': route_id})
            graph[s2].append({'to': s1, 'weight': weight, 'route_id': route_id})
    return graph

def dijkstra(graph, start_id, target_id):
    distances = {node: float('inf') for node in graph}
    previous = {node: None for node in graph}
    distances[start_id] = 0
    pq = [(0, start_id)]
    while pq:
        cur_dist, cur_node = heapq.heappop(pq)
        if cur_dist > distances[cur_node]:
            continue
        if cur_node == target_id:
            break
        for edge in graph.get(cur_node, []):
            new_dist = cur_dist + edge['weight']
            if new_dist < distances[edge['to']]:
                distances[edge['to']] = new_dist
                previous[edge['to']] = cur_node
                heapq.heappush(pq, (new_dist, edge['to']))
    path = []
    cur = target_id
    while cur:
        path.insert(0, cur)
        cur = previous[cur]
    total = distances[target_id] if distances[target_id] != float('inf') else 0
    return {'path': path, 'total_distance': total, 'node_count': len(path)}

def main():
    print("VERIFIKASI AKURASI DIJKSTRA (PYTHON)")
    stop_map = fetch_stops()
    routes = fetch_routes_with_stops()
    graph = build_graph(stop_map, routes)
    scenarios = [
        ('P00017', 'H00251P', 'Blok M → Tosari'), ('H00240P', 'B05492P', 'Tanjung Priok → St. MRT Lebak Bulus'), ('H00003P', 'H00178P', 'Ancol → Pondok Indah')
    ]
    for start, target, label in scenarios:
        t0 = time.time()
        res = dijkstra(graph, start, target)
        ms = (time.time() - t0) * 1000
        print(f"{label}: {res['total_distance']/1000:.2f} km, {res['node_count']} halte, {ms:.2f} ms")
if __name__ == "__main__":
    main()













def print_route_details(graph, start_id, target_id, stop_map):
    result = dijkstra(graph, start_id, target_id)
    path = result['path']
    print("No | Nama Halte | Jarak ke Sebelumnya (km) | Kumulatif (km)")
    cum = 0
    for i, sid in enumerate(path):
        name = stop_map.get(sid, {}).get('name', sid)
        if i == 0:
            print(f"{i+1} | {name[:35]} | START | {cum:.2f}")
        else:
            prev = path[i-1]
            lat1, lon1 = stop_map[prev]['lat'], stop_map[prev]['lon']
            lat2, lon2 = stop_map[sid]['lat'], stop_map[sid]['lon']
            seg = haversine(lat1, lon1, lat2, lon2) / 1000
            cum += seg
            print(f"{i+1} | {name[:35]} | {seg:8.2f} | {cum:8.2f}")
    print(f"Total: {cum:.2f} km | {len(path)} halte")

import random

def main():

    stop_map = fetch_stops()
    routes = fetch_routes_with_stops()
    graph = build_graph(stop_map, routes)
    stop_ids = list(stop_map.keys())
    random.seed(42)
    selected = random.sample(stop_ids, min(100, len(stop_ids)))
    pairs = []
    while len(pairs) < 10:
        a, b = random.choice(selected), random.choice(selected)
        if a != b and (a,b) not in pairs:
            pairs.append((a,b))
    for i, (start, target) in enumerate(pairs, 1):
        t0 = time.time()
        res = dijkstra(graph, start, target)
        ms = (time.time() - t0) * 1000
        jarak_km = res['total_distance'] / 1000
        print(f"{i}. {stop_map[start]['name']} → {stop_map[target]['name']}: {jarak_km:.2f} km, {res['node_count']} halte, {ms:.2f} ms")

if __name__ == "__main__":
